    69→/** Marks the user as handled — SyncMe skips these emails forever after. */
    70→function loadJson<T>(key: string, fallback: T): T {
    71→  if (typeof window === "undefined") return fallback;
    72→  try {
    73→    const raw = window.localStorage.getItem(key);
    74→    return raw ? (JSON.parse(raw) as T) : fallback;
    75→  } catch {
    76→    return fallback;
    77→  }
    78→}
    79→
    80→function saveJson(key: string, value: unknown): void {
    81→  try {
    82→    window.localStorage.setItem(key, JSON.stringify(value));
    83→  } catch {
    84→    // Storage blocked or full — persistence is best effort, sync still works.
    85→  }
    86→}

    88→/**
    89→ * Union merge: fresh tasks overwrite their previous version, tasks that were
    90→ * extracted earlier are kept even if this sync missed them, and done tasks
    91→ * never come back. Stale entries (due more than 14 days ago) are pruned.
    92→ */
    93→function mergeTasks(prev: SyncTask[], next: SyncTask[], done: Set<string>): SyncTask[] {
    94→  const byId = new Map<string, SyncTask>();
    95→  for (const t of prev) if (!done.has(t.id)) byId.set(t.id, t);
    96→  for (const t of next) if (!done.has(t.id)) byId.set(t.id, t);
    97→  const now = Date.now();
    98→  return [...byId.values()]
    99→    .filter((t) => {
   100→      const due = Date.parse(t.dueDate);
   101→      return Number.isNaN(due) || due > now - 14 * 86_400_000;
   102→    })
   103→    .slice(0, 80);
   104→}
   105→
   106→function mergeEmails(prev: RawEmail[], next: RawEmail[]): RawEmail[] {
   107→  const byId = new Map<string, RawEmail>();
   108→  for (const e of prev) byId.set(e.id, e);
   109→  for (const e of next) byId.set(e.id, e);
   110→  return [...byId.values()].sort((a, b) => Date.parse(b.date) - Date.parse(a.date)).slice(0, 50);
   111→}

   113→function formatDue(dueDate: string): string {
   114→  if (dueDate === "unknown") return "No date detected";
   115→  const d = new Date(dueDate);
   116→  if (Number.isNaN(d.getTime())) return "No date detected";
   117→  return format(d, "EEE, d MMM · HH:mm");
   118→}

   120→function initialsOf(name: string): string {
   121→  const parts = name.trim().split(/\s+/);
   122→  const chars = parts.map((p) => p[0]?.toUpperCase() ?? "").join("");
   123→  return chars.slice(0, 2) || "S";
   124→}

   190→export function DashboardClient({ user, isDemo }: { user: SyncMeUser; isDemo: boolean }) {
   191→  const [gmailTasks, setGmailTasks] = useState<SyncTask[]>(isDemo ? DEMO_TASKS : []);
   192→  const [moodleTasks, setMoodleTasks] = useState<SyncTask[]>([]);
   193→  const [emails, setEmails] = useState<RawEmail[]>(isDemo ? DEMO_EMAILS : []);
   194→  const [syncing, setSyncing] = useState(!isDemo);
   195→  const [lastSync, setLastSync] = useState<Date | null>(null);
   196→  const [connected, setConnected] = useState(true);
   197→  const [aiStatus, setAiStatus] = useState<"ai" | "heuristic" | null>(null);
   198→  const [engine, setEngine] = useState<"gemini" | "zai" | "rule" | null>(null);
   199→  const [doneIds, setDoneIds] = useState<Set<string>>(new Set());
   200→  const [moodleConnected, setMoodleConnected] = useState(false);
   201→  const [filter, setFilter] = useState<"All" | TaskCategory>("All");
   202→  const [moodleOpen, setMoodleOpen] = useState(false);
   203→  const [moodleUrl, setMoodleUrl] = useState("");
   204→  const [moodleConnecting, setMoodleConnecting] = useState(false);
   205→
   206→  /** Ref mirror of doneIds so the 30s polling merge never resurrects done tasks. */
   207→  const doneRef = useRef<Set<string>>(new Set());
   208→
   209→  // ── Restore the persisted session state before the first sync fires ───────
   210→  useEffect(() => {
   211→    if (isDemo) return;
   212→    const savedTasks = loadJson<SyncTask[]>(GMAIL_TASKS_KEY, []);
   213→    const savedMoodle = loadJson<SyncTask[]>(MOODLE_TASKS_KEY, []);
   214→    const savedEmails = loadJson<RawEmail[]>(EMAILS_KEY, []);
   215→    const savedDone = loadJson<string[]>(DONE_KEY, []);
   216→    doneRef.current = new Set(savedDone);
   217→    setDoneIds(new Set(savedDone));
   218→    setGmailTasks((prev) => (prev.length === 0 ? savedTasks : prev));
   219→    setMoodleTasks((prev) => (prev.length === 0 ? savedMoodle : prev));
   220→    setEmails((prev) => (prev.length === 0 ? savedEmails : prev));
   221→    if (savedMoodle.length > 0) setMoodleConnected(true);
   222→  }, [isDemo]);
   223→
   224→  // ── Persist on every change (best effort, browser only — zero servers) ────
   225→  useEffect(() => {
   226→    if (!isDemo) saveJson(GMAIL_TASKS_KEY, gmailTasks);
   227→  }, [gmailTasks, isDemo]);
   228→  useEffect(() => {
   229→    if (!isDemo) saveJson(MOODLE_TASKS_KEY, moodleTasks);
   230→  }, [moodleTasks, isDemo]);
   231→  useEffect(() => {
   232→    if (!isDemo) saveJson(EMAILS_KEY, emails);
   233→  }, [emails, isDemo]);
   234→  useEffect(() => {
   235→    if (!isDemo) saveJson(DONE_KEY, [...doneIds]);
   236→  }, [doneIds, isDemo]);
   237→
   238→  // ── Gmail sync (also runs on load and polls every 30s) ────────────────────
   239→  const syncNow = useCallback(
   240→    async (silent = false) => {
   241→      if (isDemo) {
   242→        if (!silent) {
   243→          toast.info("Demo sandbox", {
   244→            description: "Simulated sync — connect real Gmail from the landing page to run the live pipeline.",
   245→          });
   246→        }
   247→        setLastSync(new Date());
   248→        return;
   249→      }
   250→      setSyncing(true);
   251→      try {
   252→        const res = await fetch("/api/gmail/sync", { cache: "no-store" });
   253→        const data = (await res.json()) as GmailSyncResponse;
   254→        if (res.status === 401 || !data.ok) {
   255→          if (res.status === 401) {
   256→            setConnected(false);
   257→            if (!silent) {
   258→              toast.error("Gmail not connected", {
   259→                description: "Your session expired or is missing. Reconnect below — no consent screen needed.",
   260→              });
   261→            }
   262→          } else if (!silent) {
   263→            toast.error("Sync failed", { description: data.error ?? `HTTP ${res.status}` });
   264→          }
   265→          return;
   266→        }
   267→        // Merge instead of replace: previously synced tasks never fade away.
   268→        setGmailTasks((prev) => mergeTasks(prev, data.tasks, doneRef.current));
   269→        setEmails((prev) => mergeEmails(prev, data.emails));
   270→        setAiStatus(data.aiStatus);
   271→        setEngine(data.engine ?? null);
   272→        setConnected(true);
   273→        setLastSync(new Date());
   274→        if (!silent) {
   275→          toast.success(
   276→            `${data.tasks.length} task${data.tasks.length === 1 ? "" : "s"} extracted`,
   277→            {
   278→              description:
   279→                data.aiStatus === "heuristic"
   280→                  ? "AI was unavailable — used the rule-based extractor."
   281→                  : `${data.engine === "gemini" ? "Gemini" : "GLM"} scanned ${data.emails.length} university email${data.emails.length === 1 ? "" : "s"} (last 30 days).`,
   282→            },
   283→          );
   284→        }
   285→      } catch (err) {
   286→        if (!silent) toast.error("Sync failed", { description: String(err) });
   287→      } finally {
   288→        setSyncing(false);
   289→      }
   290→    },
   291→    [isDemo],
   292→  );
   293→
   294→  useEffect(() => {
   295→    if (isDemo) return;
   296→    void syncNow(true);
   297→    const timer = setInterval(() => void syncNow(true), POLL_INTERVAL_MS);
   298→    return () => clearInterval(timer);
   299→  }, [isDemo, syncNow]);
   300→
   301→  // ── Moodle .ics connect ────────────────────────────────────────────────────
   302→  const connectMoodle = useCallback(
   303→    async (url: string, silent = false) => {
   304→      if (!/^https?:\/\/.+/i.test(url)) {
   305→        if (!silent) toast.error("Invalid calendar URL", { description: "Paste a full http(s) .ics export URL." });
   306→        return;
   307→      }
   308→      setMoodleConnecting(true);
   309→      try {
   310→        const res = await fetch("/api/moodle/sync", {
   311→          method: "POST",
   312→          headers: { "Content-Type": "application/json" },
   313→          body: JSON.stringify({ icsUrl: url }),
   314→        });
   315→        const data = (await res.json()) as MoodleSyncResponse;
   316→        if (!res.ok || !data.ok) {
   317→          if (!silent) toast.error("Moodle sync failed", { description: data.error ?? `HTTP ${res.status}` });
   318→          return;
   319→        }
   320→        setMoodleTasks((prev) => mergeTasks(prev, data.tasks, doneRef.current));
   321→        setMoodleConnected(true);
   322→        window.localStorage.setItem(MOODLE_URL_KEY, url);
   323→        if (!silent) {
   324→          toast.success(`${data.tasks.length} Moodle event${data.tasks.length === 1 ? "" : "s"} imported`, {
   325→            description: "Deadlines from your e-learning calendar now appear alongside Gmail tasks.",
   326→          });
   327→          setMoodleOpen(false);
   328→        }
   329→      } catch (err) {
   330→        if (!silent) toast.error("Moodle sync failed", { description: String(err) });
   331→      } finally {
   332→        setMoodleConnecting(false);
   333→      }
   334→    },
   335→    [],
   336→  );
   337→
   338→  useEffect(() => {
   339→    if (isDemo) return;
   340→    const saved = window.localStorage.getItem(MOODLE_URL_KEY);
   341→    if (saved) {
   342→      setMoodleUrl(saved);
   343→      void connectMoodle(saved, true);
   344→    }
   345→  }, [isDemo, connectMoodle]);
   346→
   347→  /** Mark as done: hidden now, skipped by every future sync, remembered forever. */
   348→  const markTaskDone = useCallback((task: SyncTask) => {
   349→    const next = new Set(doneRef.current);
   350→    next.add(task.id);
   351→    doneRef.current = next;
   352→    setDoneIds(next);
   353→    setGmailTasks((prev) => prev.filter((t) => t.id !== task.id));
   354→    setMoodleTasks((prev) => prev.filter((t) => t.id !== task.id));
   355→    toast.success("Marked as done", {
   356→      description: `SyncMe will skip "${task.title}" from now on.`,
   357→    });
   358→  }, []);
   359→
   360→  const logout = useCallback(async () => {
   361→    try {
   362→      await fetch("/api/auth/logout", { method: "POST" });

   364→      window.location.href = "/";
   365→    }
   366→  }, []);

   371→    return [...merged].sort((a, b) => {

   375→      const vb = Number.isNaN(tb) ? Number.POSITIVE_INFINITY : tb;

   380→  const filteredTasks = useMemo(
   381→    () => (filter === "All" ? allTasks : allTasks.filter((t) => t.category === filter)),

   384→

   393→  const showSkeletons = syncing && allTasks.length === 0;

   401→          <div className="flex items-center gap-2 sm:gap-3">

   403→              <Badge variant="outline" className="gap-1.5 border-primary/40 bg-primary/10 text-primary">

   411→                <AvatarFallback className="bg-secondary text-xs font-semibold">
   412→                  {initialsOf(user.name)}
   413→                </AvatarFallback>
   414→              </Avatar>

   429→        {/* ── Demo banner ──────────────────────────────────────────────────── */}

   432→            <FlaskConical className="h-4 w-4 shrink-0 text-primary" />
