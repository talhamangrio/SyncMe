    74→    74→function loadJson<T>(key: string, fallback: T): T {
    75→    75→  if (typeof window === "undefined") return fallback;
    76→    76→  try {
    77→    77→    const raw = window.localStorage.getItem(key);
    78→    78→    return raw ? (JSON.parse(raw) as T) : fallback;
    79→    79→  } catch {
    80→    80→    return fallback;
    81→    81→  }
    82→    82→}

    84→    84→function saveJson(key: string, value: unknown): void {
    85→    85→  try {
    86→    86→    window.localStorage.setItem(key, JSON.stringify(value));
    87→    87→  } catch {
    88→    88→    // Storage blocked or full — persistence is best effort, sync still works.
    89→    89→  }
    90→    90→}

    92→    92→/**
    93→    93→ * Union merge: fresh tasks overwrite their previous version, tasks that were
    94→    94→ * extracted earlier are kept even if this sync missed them, and done tasks
    95→    95→ * never come back. Stale entries (due more than 14 days ago) are pruned.
    96→    96→ */
    97→    97→function mergeTasks(prev: SyncTask[], next: SyncTask[], done: Set<string>): SyncTask[] {
    98→    98→  const byId = new Map<string, SyncTask>();
    99→    99→  for (const t of prev) if (!done.has(t.id)) byId.set(t.id, t);
   100→   100→  for (const t of next) if (!done.has(t.id)) byId.set(t.id, t);
   101→   101→  const now = Date.now();
   102→   102→  return [...byId.values()]
   103→   103→    .filter((t) => {
   104→   104→      const due = Date.parse(t.dueDate);
   105→   105→      return Number.isNaN(due) || due > now - 14 * 86_400_000;
   106→   106→    })
   107→   107→    .slice(0, 80);
   108→   108→}

   110→   110→function mergeEmails(prev: RawEmail[], next: RawEmail[]): RawEmail[] {
   111→   111→  const byId = new Map<string, RawEmail>();
   112→   112→  for (const e of prev) byId.set(e.id, e);
   113→   113→  for (const e of next) byId.set(e.id, e);
   114→   114→  return [...byId.values()].sort((a, b) => Date.parse(b.date) - Date.parse(a.date)).slice(0, 50);
   115→   115→}

   117→   117→function formatDue(dueDate: string): string {
   118→   118→  if (dueDate === "unknown") return "No date detected";
   119→   119→  const d = new Date(dueDate);
   120→   120→  if (Number.isNaN(d.getTime())) return "No date detected";
   121→   121→  return format(d, "EEE, d MMM · HH:mm");
   122→   122→}

   124→   124→function initialsOf(name: string): string {
   125→   125→  const parts = name.trim().split(/\s+/);
   126→   126→  const chars = parts.map((p) => p[0]?.toUpperCase() ?? "").join("");
   127→   127→  return chars.slice(0, 2) || "S";
   128→   128→}

   224→   224→export function DashboardClient({ user, isDemo }: { user: SyncMeUser; isDemo: boolean }) {

   226→   226→  const [moodleTasks, setMoodleTasks] = useState<SyncTask[]>([]);
   227→   227→  const [emails, setEmails] = useState<RawEmail[]>(isDemo ? DEMO_EMAILS : []);
   228→   228→  const [syncing, setSyncing] = useState(!isDemo);
   229→   229→  const [lastSync, setLastSync] = useState<Date | null>(null);
   230→   230→  const [connected, setConnected] = useState(true);
   231→   231→  const [aiStatus, setAiStatus] = useState<"ai" | "heuristic" | null>(null);
   232→   232→  const [engine, setEngine] = useState<"gemini" | "zai" | "rule" | null>(null);
   233→   233→  const [doneIds, setDoneIds] = useState<Set<string>>(new Set());
   234→   234→  const [moodleConnected, setMoodleConnected] = useState(false);
   235→   235→  const [filter, setFilter] = useState<"All" | TaskCategory>("All");
   236→   236→  const [moodleOpen, setMoodleOpen] = useState(false);
   237→   237→  const [moodleUrl, setMoodleUrl] = useState("");
   238→   238→  const [moodleConnecting, setMoodleConnecting] = useState(false);
   239→   239→
   240→   240→  /** Ref mirror of doneIds so the 30s polling merge never resurrects done tasks. */
   241→   241→  const doneRef = useRef<Set<string>>(new Set());
   242→   242→
   243→   243→  // ── Restore the persisted session state before the first sync fires ───────
   244→   244→  useEffect(() => {
   245→   245→    if (isDemo) return;
   246→   246→    const savedTasks = loadJson<SyncTask[]>(GMAIL_TASKS_KEY, []);
   247→   247→    const savedMoodle = loadJson<SyncTask[]>(MOODLE_TASKS_KEY, []);
   248→   248→    const savedEmails = loadJson<RawEmail[]>(EMAILS_KEY, []);
   249→   249→    const savedDone = loadJson<string[]>(DONE_KEY, []);
   250→   250→    doneRef.current = new Set(savedDone);
   251→   251→    setDoneIds(new Set(savedDone));
   252→   252→    setGmailTasks((prev) => (prev.length === 0 ? savedTasks : prev));
   253→   253→    setMoodleTasks((prev) => (prev.length === 0 ? savedMoodle : prev));
   254→   254→    setEmails((prev) => (prev.length === 0 ? savedEmails : prev));
   255→   255→    if (savedMoodle.length > 0) setMoodleConnected(true);
   256→   256→  }, [isDemo]);
   257→   257→
   258→   258→  // ── Persist on every change (best effort, browser only — zero servers) ────
   259→   259→  useEffect(() => {
   260→   260→    if (!isDemo) saveJson(GMAIL_TASKS_KEY, gmailTasks);
   261→   261→  }, [gmailTasks, isDemo]);
   262→   262→  useEffect(() => {
   263→   263→    if (!isDemo) saveJson(MOODLE_TASKS_KEY, moodleTasks);
   264→   264→  }, [moodleTasks, isDemo]);
   265→   265→  useEffect(() => {
   266→   266→    if (!isDemo) saveJson(EMAILS_KEY, emails);
   267→   267→  }, [emails, isDemo]);
   268→   268→  useEffect(() => {
   269→   269→    if (!isDemo) saveJson(DONE_KEY, [...doneIds]);
   270→   270→  }, [doneIds, isDemo]);
   271→   271→
   272→   272→  // ── Gmail sync (also runs on load and polls every 30s) ────────────────────
   273→   273→  const syncNow = useCallback(
   274→   274→    async (silent = false) => {
   275→   275→      if (isDemo) {
   276→   276→        if (!silent) {
   277→   277→          toast.info("Demo sandbox", {
   278→   278→            description: "Simulated sync — connect real Gmail from the landing page to run the live pipeline.",
   279→   279→          });
   280→   280→        }
   281→   281→        setLastSync(new Date());
   282→   282→        return;
   283→   283→      }
   284→   284→      setSyncing(true);
   285→   285→      try {
   286→   286→        const res = await fetch("/api/gmail/sync", { cache: "no-store" });
   287→   287→        const data = (await res.json()) as GmailSyncResponse;
   288→   288→        if (res.status === 401 || !data.ok) {
   289→   289→          if (res.status === 401) {
   290→   290→            setConnected(false);
   291→   291→            if (!silent) {
   292→   292→              toast.error("Gmail not connected", {
   293→   293→                description: "Your session expired or is missing. Reconnect below — no consent screen needed.",
   294→   294→              });
   295→   295→            }
   296→   296→          } else if (!silent) {
   297→   297→            toast.error("Sync failed", { description: data.error ?? `HTTP ${res.status}` });
   298→   298→          }
   299→   299→          return;
   300→   300→        }
   301→   301→        // Merge instead of replace: previously synced tasks never fade away.
   302→   302→        setGmailTasks((prev) => mergeTasks(prev, data.tasks, doneRef.current));
   303→   303→        setEmails((prev) => mergeEmails(prev, data.emails));
   304→   304→        setAiStatus(data.aiStatus);
   305→   305→        setEngine(data.engine ?? null);
   306→   306→        setConnected(true);
   307→   307→        setLastSync(new Date());
   308→   308→        if (!silent) {
   309→   309→          toast.success(
   310→   310→            `${data.tasks.length} task${data.tasks.length === 1 ? "" : "s"} extracted`,
   311→   311→            {
   312→   312→              description:
   313→   313→                data.aiStatus === "heuristic"
   314→   314→                  ? "AI was unavailable — used the rule-based extractor."

   317→   317→          );
   318→   318→        }

   321→   321→      } finally {
   322→   322→        setSyncing(false);
   323→   323→      }

   363→   363→      } catch (err) {

   366→   366→        setMoodleConnecting(false);
   367→   367→      }
   368→   368→    },

   370→   370→  );
   371→   371→
   372→   372→  useEffect(() => {
   373→   373→    if (isDemo) return;
   374→   374→    const saved = window.localStorage.getItem(MOODLE_URL_KEY);
   375→   375→    if (saved) {
   376→   376→      setMoodleUrl(saved);
   377→   377→      void connectMoodle(saved, true);
   378→   378→    }
   379→   379→  }, [isDemo, connectMoodle]);

   381→   381→  /** Mark as done: hidden now, skipped by every future sync, remembered forever. */

   384→   384→    next.add(task.id);
   385→   385→    doneRef.current = next;
   386→   386→    setDoneIds(next);
   387→   387→    setGmailTasks((prev) => prev.filter((t) => t.id !== task.id));
   388→   388→    setMoodleTasks((prev) => prev.filter((t) => t.id !== task.id));
   389→   389→    toast.success("Marked as done", {
   390→   390→      description: `SyncMe will skip "${task.title}" from now on.`,
   391→   391→    });
   392→   392→  }, []);
   393→   393→
   394→   394→  const logout = useCallback(async () => {
   395→   395→    try {
   396→   396→      await fetch("/api/auth/logout", { method: "POST" });
   397→   397→    } finally {
   398→   398→      window.location.href = "/";
   399→   399→    }
   400→   400→  }, []);
   401→   401→
   402→   402→  // ── Derived data ───────────────────────────────────────────────────────────
   403→   403→  const allTasks = useMemo(() => {
   404→   404→    const merged = [...gmailTasks, ...moodleTasks];
   405→   405→    return [...merged].sort((a, b) => {
   406→   406→      const ta = Date.parse(a.dueDate);
   407→   407→      const tb = Date.parse(b.dueDate);
   408→   408→      const va = Number.isNaN(ta) ? Number.POSITIVE_INFINITY : ta;
   409→   409→      const vb = Number.isNaN(tb) ? Number.POSITIVE_INFINITY : tb;
   410→   410→      return va - vb;
   411→   411→    });
   412→   412→  }, [gmailTasks, moodleTasks]);
   413→   413→
   414→   414→  const filteredTasks = useMemo(() => {
   415→   415→    const list = filter === "All" ? allTasks : allTasks.filter((t) => t.category === filter);
   416→   416→    const dueMs = (x: SyncTask) => {
   417→   417→      const d = Date.parse(x.dueDate);
   418→   418→      return Number.isNaN(d) ? Number.POSITIVE_INFINITY : d;
   419→   419→    };
   420→   420→    // Priority first (High → Medium → Low), soonest deadline inside each group.
   421→   421→    return [...list].sort(
   422→   422→      (a, b) => PRIORITY_RANK[a.priority] - PRIORITY_RANK[b.priority] || dueMs(a) - dueMs(b),
   423→   423→    );
   424→   424→  }, [allTasks, filter]);
   425→   425→
