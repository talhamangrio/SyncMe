    76→/** Marks the user as handled — SyncMe skips these emails forever after. */
    77→function loadJson<T>(key: string, fallback: T): T {
    78→  if (typeof window === "undefined") return fallback;
    79→  try {
    80→    const raw = window.localStorage.getItem(key);
    81→    return raw ? (JSON.parse(raw) as T) : fallback;
    82→  } catch {
    83→    return fallback;
    84→  }
    85→}
    86→
    87→function saveJson(key: string, value: unknown): void {
    88→  try {
    89→    window.localStorage.setItem(key, JSON.stringify(value));
    90→  } catch {
    91→    // Storage blocked or full — persistence is best effort, sync still works.
    92→  }
    93→}

    95→/**
    96→ * Union merge: fresh tasks overwrite their previous version, tasks that were
    97→ * extracted earlier are kept even if this sync missed them, and done tasks
    98→ * never come back. Expired entries (dated strictly before today) are pruned —
    99→ * the dashboard can never resurrect the past.
   100→ */
   101→function mergeTasks(prev: SyncTask[], next: SyncTask[], done: Set<string>): SyncTask[] {
   102→  const byId = new Map<string, SyncTask>();
   103→  for (const t of prev) if (!done.has(t.id)) byId.set(t.id, t);
   104→  for (const t of next) if (!done.has(t.id)) byId.set(t.id, t);
   105→  const today = budapestTodayKey();
   106→  return [...byId.values()]
   107→    .filter((t) => {
   108→      const day = dayKeyOf(t.dueDate);
   109→      return day === null || day >= today;
   110→    })
   111→    .slice(0, 80);
   112→}
   113→
   114→function mergeEmails(prev: RawEmail[], next: RawEmail[]): RawEmail[] {
   115→  const byId = new Map<string, RawEmail>();
   116→  for (const e of prev) byId.set(e.id, e);
   117→  for (const e of next) byId.set(e.id, e);
   118→  return [...byId.values()].sort((a, b) => Date.parse(b.date) - Date.parse(a.date)).slice(0, 50);
   119→}

   121→function formatDue(dueDate: string): string {
   122→  if (dueDate === "unknown") return "No date detected";
   123→  const d = new Date(dueDate);
   124→  if (Number.isNaN(d.getTime())) return "No date detected";
   125→  return format(d, "EEE, d MMM · HH:mm");
   126→}

   128→/** Sort key: normalized Budapest ISO parsed as ms; undated tasks sink to the end. */
   129→function dueMsOf(task: SyncTask): number {
   130→  const normalized = normalizeDueDate(task.dueDate);
   131→  return normalized === "unknown" ? Number.POSITIVE_INFINITY : Date.parse(normalized);
   132→}

   134→function initialsOf(name: string): string {
   135→  const parts = name.trim().split(/\s+/);
   136→  const chars = parts.map((p) => p[0]?.toUpperCase() ?? "").join("");
   137→  return chars.slice(0, 2) || "S";
   138→}

   140→function greetingOf(d: Date): string {
   141→  const h = d.getHours();
   142→  if (h < 12) return "Good morning";
   143→  if (h < 18) return "Good afternoon";
   144→  return "Good evening";
   145→}

   147→/** Human label for a `YYYY-MM-DD` day key: "Tomorrow • Nov 25", "Thu • Nov 26", … */
   148→function dayLabel(dayKey: string, todayKey: string): string {
   149→  const d = new Date(`${dayKey}T12:00:00Z`);
   150→  const sameYear = d.getUTCFullYear() === new Date().getUTCFullYear();
   151→  const datePart = `${format(d, "MMM d")}${sameYear ? "" : `, ${format(d, "yyyy")}`}`;
   152→  if (dayKey === addDaysKey(todayKey, 1)) return `Tomorrow • ${datePart}`;
   153→  return `${format(d, "EEE")} • ${datePart}`;
   154→}

   167→  task,
   168→  index,

   182→    <motion.li
   183→      initial={{ opacity: 0, x: -8 }}

   235→          type="button"

   248→export function DashboardClient({ user, isDemo }: { user: SyncMeUser; isDemo: boolean }) {
   249→  const [gmailTasks, setGmailTasks] = useState<SyncTask[]>(isDemo ? DEMO_TASKS : []);
   250→  const [moodleTasks, setMoodleTasks] = useState<SyncTask[]>([]);
   251→  const [emails, setEmails] = useState<RawEmail[]>(isDemo ? DEMO_EMAILS : []);
   252→  const [syncing, setSyncing] = useState(!isDemo);
   253→  const [lastSync, setLastSync] = useState<Date | null>(null);
   254→  const [connected, setConnected] = useState(true);
   255→  const [aiStatus, setAiStatus] = useState<"ai" | "heuristic" | null>(null);
   256→  const [engine, setEngine] = useState<"gemini" | "zai" | "rule" | null>(null);
   257→  const [doneIds, setDoneIds] = useState<Set<string>>(new Set());
   258→  const [moodleConnected, setMoodleConnected] = useState(false);
   259→  const [filter, setFilter] = useState<"All" | TaskCategory>("All");
   260→  const [moodleOpen, setMoodleOpen] = useState(false);
   261→  const [moodleUrl, setMoodleUrl] = useState("");
   262→  const [moodleConnecting, setMoodleConnecting] = useState(false);
   263→  const [todayKey, setTodayKey] = useState<string>(() => budapestTodayKey());
   264→
   265→  /** Ref mirror of doneIds so the 30s polling merge never resurrects done tasks. */
   266→  const doneRef = useRef<Set<string>>(new Set());
   267→
   268→  // ── Keep the Budapest "today" key fresh (handles midnight rollover) ────────
   269→  useEffect(() => {
   270→    const update = () => setTodayKey(budapestTodayKey());
   271→    update();
   272→    const timer = setInterval(update, 60_000);
   273→    return () => clearInterval(timer);
   274→  }, []);
   275→
   276→  // ── Restore the persisted session state before the first sync fires ───────
   277→  useEffect(() => {
   278→    if (isDemo) return;
   279→    const savedTasks = loadJson<SyncTask[]>(GMAIL_TASKS_KEY, []);
   280→    const savedMoodle = loadJson<SyncTask[]>(MOODLE_TASKS_KEY, []);
   281→    const savedEmails = loadJson<RawEmail[]>(EMAILS_KEY, []);
   282→    const savedDone = loadJson<string[]>(DONE_KEY, []);
   283→    doneRef.current = new Set(savedDone);
   284→    setDoneIds(new Set(savedDone));
   285→    setGmailTasks((prev) => (prev.length === 0 ? savedTasks : prev));
   286→    setMoodleTasks((prev) => (prev.length === 0 ? savedMoodle : prev));
   287→    setEmails((prev) => (prev.length === 0 ? savedEmails : prev));
   288→    if (savedMoodle.length > 0) setMoodleConnected(true);
   289→  }, [isDemo]);
   290→
   291→  // ── Persist on every change (best effort, browser only — zero servers) ────
   292→  useEffect(() => {
   293→    if (!isDemo) saveJson(GMAIL_TASKS_KEY, gmailTasks);
   294→  }, [gmailTasks, isDemo]);
   295→  useEffect(() => {
   296→    if (!isDemo) saveJson(MOODLE_TASKS_KEY, moodleTasks);
   297→  }, [moodleTasks, isDemo]);
   298→  useEffect(() => {
   299→    if (!isDemo) saveJson(EMAILS_KEY, emails);
   300→  }, [emails, isDemo]);
   301→  useEffect(() => {
   302→    if (!isDemo) saveJson(DONE_KEY, [...doneIds]);
   303→  }, [doneIds, isDemo]);
   304→
   305→  // ── Gmail sync (also runs on load and polls every 30s) ────────────────────
   306→  const syncNow = useCallback(
   307→    async (silent = false) => {
   308→      if (isDemo) {
   309→        if (!silent) {
   310→          toast.info("Demo sandbox", {
   311→            description: "Simulated sync — connect real Gmail from the landing page to run the live pipeline.",
   312→          });
   313→        }
   314→        setLastSync(new Date());
   315→        return;
   316→      }
   317→      setSyncing(true);
   318→      try {
   319→        const res = await fetch("/api/gmail/sync", { cache: "no-store" });
   320→        const data = (await res.json()) as GmailSyncResponse;
   321→        if (res.status === 401 || !data.ok) {
   322→          if (res.status === 401) {
   323→            setConnected(false);
   324→            if (!silent) {
   325→              toast.error("Gmail not connected", {
   326→                description: "Your session expired or is missing. Reconnect below — no consent screen needed.",
   327→              });
   328→            }
   329→          } else if (!silent) {
   330→            toast.error("Sync failed", { description: data.error ?? `HTTP ${res.status}` });
   331→          }
   332→          return;
   333→        }
   334→        // Merge instead of replace: previously synced tasks never fade away.
   335→        setGmailTasks((prev) => mergeTasks(prev, data.tasks, doneRef.current));
   336→        setEmails((prev) => mergeEmails(prev, data.emails));
   337→        setAiStatus(data.aiStatus);
   338→        setEngine(data.engine ?? null);
   339→        setConnected(true);
   340→        setLastSync(new Date());
   341→        if (!silent) {
   342→          toast.success(
   343→            `${data.tasks.length} task${data.tasks.length === 1 ? "" : "s"} extracted`,
   344→            {
   345→              description:
   346→                data.aiStatus === "heuristic"
   347→                  ? "AI was unavailable — used the rule-based extractor."
   348→                  : `${data.engine === "gemini" ? "Gemini" : "GLM"} scanned ${data.emails.length} university email${data.emails.length === 1 ? "" : "s"} (last 30 days).`,
   349→            },
   350→          );
   351→        }
   352→      } catch (err) {
   353→        if (!silent) toast.error("Sync failed", { description: String(err) });
   354→      } finally {
   355→        setSyncing(false);
   356→      }
   357→    },
   358→    [isDemo],
   359→  );

   372→        if (!silent) toast.error("Invalid calendar URL", { description: "Paste a full http(s) .ics export URL." });
   373→        return;
   374→      }

   376→      try {
   377→        const res = await fetch("/api/moodle/sync", {
   378→          method: "POST",
   379→          headers: { "Content-Type": "application/json" },
   380→          body: JSON.stringify({ icsUrl: url }),
   381→        });
   382→        const data = (await res.json()) as MoodleSyncResponse;
   383→        if (!res.ok || !data.ok) {
   384→          if (!silent) toast.error("Moodle sync failed", { description: data.error ?? `HTTP ${res.status}` });
   385→          return;
   386→        }
   387→        setMoodleTasks((prev) => mergeTasks(prev, data.tasks, doneRef.current));
   388→        setMoodleConnected(true);
   389→        window.localStorage.setItem(MOODLE_URL_KEY, url);
   390→        if (!silent) {
   391→          toast.success(`${data.tasks.length} Moodle event${data.tasks.length === 1 ? "" : "s"} imported`, {
   392→            description: "Deadlines from your e-learning calendar now appear alongside Gmail tasks.",
   393→          });
   394→          setMoodleOpen(false);
   395→        }
   396→      } catch (err) {
   397→        if (!silent) toast.error("Moodle sync failed", { description: String(err) });
   398→      } finally {
   399→        setMoodleConnecting(false);
   400→      }
   401→    },
   402→    [],
   403→  );
   404→
   405→  useEffect(() => {
   406→    if (isDemo) return;
   407→    const saved = window.localStorage.getItem(MOODLE_URL_KEY);
   408→    if (saved) {
   409→      setMoodleUrl(saved);
   410→      void connectMoodle(saved, true);
   411→    }
   412→  }, [isDemo, connectMoodle]);

   414→  /** Mark as done: hidden now, skipped by every future sync, remembered forever. */
   415→  const markTaskDone = useCallback((task: SyncTask) => {
   416→    const next = new Set(doneRef.current);
   417→    next.add(task.id);
   418→    doneRef.current = next;
   419→    setDoneIds(next);
   420→    setGmailTasks((prev) => prev.filter((t) => t.id !== task.id));
   421→    setMoodleTasks((prev) => prev.filter((t) => t.id !== task.id));
   422→    toast.success("Marked as done", {
   423→      description: `SyncMe will skip "${task.title}" from now on.`,
   424→    });
   425→  }, []);
   426→
   427→  const logout = useCallback(async () => {
   428→    try {
   429→      await fetch("/api/auth/logout", { method: "POST" });
   430→    } finally {
   431→      window.location.href = "/";
   432→    }
   433→  }, []);
   434→
   435→  // ── Derived data ───────────────────────────────────────────────────────────
   436→  const allTasks = useMemo(() => {
   437→    const merged = [...gmailTasks, ...moodleTasks];
   438→    // Strict ascending chronological order — earliest date/time first, undated last.
   439→    return [...merged].sort((a, b) => dueMsOf(a) - dueMsOf(b));
   440→  }, [gmailTasks, moodleTasks]);
   441→
   442→  /** Client-side date guard: expired tasks can never render (covers demo data too). */
   443→  const visibleTasks = useMemo(() => {
   444→    const list = filter === "All" ? allTasks : allTasks.filter((t) => t.category === filter);
   445→    return list.filter((t) => {
   446→      const day = dayKeyOf(t.dueDate);
   447→      return day === null || day >= todayKey;
   448→    });
   449→  }, [allTasks, filter, todayKey]);
   450→
   451→  /** Section 1 — everything due today: High priority first, then by time. */
   452→  const focusTasks = useMemo(
   453→    () =>
   454→      visibleTasks
   455→        .filter((t) => dayKeyOf(t.dueDate) === todayKey)
   456→        .sort((a, b) => PRIORITY_RANK[a.priority] - PRIORITY_RANK[b.priority] || dueMsOf(a) - dueMsOf(b)),
   457→    [visibleTasks, todayKey],
   458→  );
   459→
   460→  /** Section 2 — everything after today, grouped by day, strictly ascending. */
   461→  const upcomingDays = useMemo(() => {
   462→    const groups = new Map<string, SyncTask[]>();
   463→    const undated: SyncTask[] = [];
   464→    for (const t of visibleTasks) {
   465→      const day = dayKeyOf(t.dueDate);
   466→      if (day === null) {
   467→        undated.push(t);
   468→        continue;
   469→      }
   470→      if (day <= todayKey) continue; // today lives in Today's Focus
   471→      const bucket = groups.get(day);
   472→      if (bucket) bucket.push(t);
   473→      else groups.set(day, [t]);
   474→    }
   475→    const days = [...groups.entries()]
   476→      .sort(([a], [b]) => a.localeCompare(b))
   477→      .map(([day, items]) => ({ key: day, items: items.sort((a, b) => dueMsOf(a) - dueMsOf(b)) }));
   478→    if (undated.length > 0) days.push({ key: "undated", items: undated });
   479→    return days;
   480→  }, [visibleTasks, todayKey]);
   481→
   482→  /** The very next chronological task — previewed when today is clear. */
   483→  const nextTask = upcomingDays[0]?.items[0] ?? null;
   484→
   485→  const conflicts = useMemo(() => findConflicts(visibleTasks), [visibleTasks]);
   486→
   487→  const categoryCounts = useMemo(() => {
   488→    const counts = new Map<"All" | TaskCategory, number>([["All", visibleTasks.length]]);
   489→    for (const c of TASK_CATEGORIES) counts.set(c, visibleTasks.filter((t) => t.category === c).length);
   490→    return counts;
   491→  }, [visibleTasks]);

   493→  const totalUpcoming = upcomingDays.reduce((sum, g) => sum + g.items.length, 0);
