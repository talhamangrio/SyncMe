    70→    70→function loadJson<T>(key: string, fallback: T): T {
    71→    71→  if (typeof window === "undefined") return fallback;
    72→    72→  try {
    73→    73→    const raw = window.localStorage.getItem(key);
    74→    74→    return raw ? (JSON.parse(raw) as T) : fallback;
    75→    75→  } catch {
    76→    76→    return fallback;
    77→    77→  }
    78→    78→}

    80→    80→function saveJson(key: string, value: unknown): void {
    81→    81→  try {
    82→    82→    window.localStorage.setItem(key, JSON.stringify(value));
    83→    83→  } catch {
    84→    84→    // Storage blocked or full — persistence is best effort, sync still works.
    85→    85→  }
    86→    86→}

    88→    88→/**
    89→    89→ * Union merge: fresh tasks overwrite their previous version, tasks that were
    90→    90→ * extracted earlier are kept even if this sync missed them, and done tasks
    91→    91→ * never come back. Stale entries (due more than 14 days ago) are pruned.
    92→    92→ */
    93→    93→function mergeTasks(prev: SyncTask[], next: SyncTask[], done: Set<string>): SyncTask[] {
    94→    94→  const byId = new Map<string, SyncTask>();
    95→    95→  for (const t of prev) if (!done.has(t.id)) byId.set(t.id, t);
    96→    96→  for (const t of next) if (!done.has(t.id)) byId.set(t.id, t);
    97→    97→  const now = Date.now();
    98→    98→  return [...byId.values()]
    99→    99→    .filter((t) => {
   100→   100→      const due = Date.parse(t.dueDate);
   101→   101→      return Number.isNaN(due) || due > now - 14 * 86_400_000;
   102→   102→    })
   103→   103→    .slice(0, 80);
   104→   104→}
   105→   105→
   106→   106→function mergeEmails(prev: RawEmail[], next: RawEmail[]): RawEmail[] {
   107→   107→  const byId = new Map<string, RawEmail>();
   108→   108→  for (const e of prev) byId.set(e.id, e);
   109→   109→  for (const e of next) byId.set(e.id, e);
   110→   110→  return [...byId.values()].sort((a, b) => Date.parse(b.date) - Date.parse(a.date)).slice(0, 50);
   111→   111→}

   113→   113→function formatDue(dueDate: string): string {
   114→   114→  if (dueDate === "unknown") return "No date detected";
   115→   115→  const d = new Date(dueDate);
   116→   116→  if (Number.isNaN(d.getTime())) return "No date detected";
   117→   117→  return format(d, "EEE, d MMM · HH:mm");
   118→   118→}

   120→   120→function initialsOf(name: string): string {
   121→   121→  const parts = name.trim().split(/\s+/);
   122→   122→  const chars = parts.map((p) => p[0]?.toUpperCase() ?? "").join("");
   123→   123→  return chars.slice(0, 2) || "S";
   124→   124→}
